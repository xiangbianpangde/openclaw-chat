## 二、OCR 集成方案（预算：2 人天）

### 2.1 问题背景

**现状：** 原方案使用 PyMuPDF + pdfplumber 解析 PDF，仅支持文本型 PDF（可直接复制文字），对扫描版 PDF（图片格式）无法识别。

**尚书省反馈：** 阶段二执行过程中，5 个 PDF 文件中有 2 个为扫描版，解析失败率 40%。

**需求：** 集成 OCR 引擎，支持扫描版 PDF 文字识别，目标准确率 >85%。

### 2.4 技术实现细节

#### 2.4.1 PDF 类型检测

```python
import fitz  # PyMuPDF

def is_scanned_pdf(pdf_path: str) -> bool:
    """
    检测 PDF 是否为扫描版
    原理：扫描版 PDF 无文本层，所有页面均为图片
    """
    doc = fitz.open(pdf_path)
    
    text_length = 0
    image_count = 0
    
    for page in doc:
        # 提取文本
        text = page.get_text()
        text_length += len(text.strip())
        
        # 统计图片数量
        image_list = page.get_images()
        image_count += len(image_list)
    
    doc.close()
    
    # 判定逻辑：
    # 1. 文本极少（<100 字符）且图片多 → 扫描版
    # 2. 文本充足 → 文本型
    if text_length < 100 and image_count > 0:
        return True
    return False
```

#### 2.4.2 PaddleOCR 集成

```python
from paddleocr import PaddleOCR
from ppstructure.table import TableStructure
import cv2
import numpy as np

class OCREngine:
    def __init__(self, use_gpu: bool = False, lang: str = 'ch'):
        """
        初始化 PaddleOCR
        :param use_gpu: 是否使用 GPU 加速
        :param lang: 识别语言（ch=中文，en=英文）
        """
        # OCR 识别引擎（轻量模型）
        self.ocr = PaddleOCR(
            use_angle_cls=True,      # 使用方向分类器
            lang=lang,
            use_gpu=use_gpu,
            det_model_dir='paddleocr/det',  # 检测模型
            rec_model_dir='paddleocr/rec',  # 识别模型
            show_log=False,
        )
        
        # 表格识别引擎
        self.table_engine = TableStructure(
            lang=lang,
            use_gpu=use_gpu,
        )
    
    def ocr_page(self, image: np.ndarray) -> dict:
        """
        对单页图片进行 OCR 识别
        :param image: 页面图片（OpenCV 格式）
        :return: {
            'text': str,              # 完整文本
            'blocks': List[dict],     # 文本块（含位置）
            'tables': List[dict],     # 表格数据
        }
        """
        # 1. 文本检测与识别
        ocr_result = self.ocr.ocr(image, cls=True)
        
        # 2. 解析 OCR 结果
        text_blocks = []
        full_text = []
        
        if ocr_result and ocr_result[0]:
            for block in ocr_result[0]:
                bbox, (text, confidence) = block
                text_blocks.append({
                    'text': text,
                    'confidence': confidence,
                    'bbox': bbox  # [x1, y1, x2, y2, x3, y3, x4, y4]
                })
                full_text.append(text)
        
        # 3. 表格识别
        table_result = self.table_engine(image)
        tables = []
        if table_result and 'html' in table_result:
            tables.append({
                'html': table_result['html'],
                'cells': table_result.get('cells', []),
                'bbox': table_result.get('bbox', [])
            })
        
        return {
            'text': '\n'.join(full_text),
            'blocks': text_blocks,
            'tables': tables
        }
    
    def parse_scanned_pdf(self, pdf_path: str) -> dict:
        """
        解析扫描版 PDF
        :param pdf_path: PDF 文件路径
        :return: {
            'pages': List[dict],  # 每页识别结果
            'full_text': str,     # 完整文本
            'all_tables': List[dict],  # 所有表格
        }
        """
        doc = fitz.open(pdf_path)
        pages_result = []
        full_text = []
        all_tables = []
        
        for page_num in range(len(doc)):
            # 1. 页面 → 图片（300 DPI）
            page = doc[page_num]
            mat = fitz.Matrix(300 / 72, 300 / 72)  # 放大到 300 DPI
            pix = page.get_pixmap(matrix=mat)
            img_data = pix.tobytes("png")
            
            # 2. OpenCV 读取图片
            nparr = np.frombuffer(img_data, np.uint8)
            image = cv2.imdecode(nparr, cv2.IMREAD_COLOR)
            
            # 3. OCR 识别
            page_result = self.ocr_page(image)
            page_result['page_num'] = page_num + 1
            pages_result.append(page_result)
            
            full_text.append(f"--- 第 {page_num + 1} 页 ---")
            full_text.append(page_result['text'])
            all_tables.extend(page_result['tables'])
        
        doc.close()
        
        return {
            'pages': pages_result,
            'full_text': '\n'.join(full_text),
            'all_tables': all_tables,
            'total_pages': len(pages_result)
        }
```

## 三、研报格式适配方案（预算：3 人天）

### 3.1 问题背景

**现状：** 原方案针对标准财报（资产负债表、利润表、现金流量表）设计提取规则，无法处理行业研报的非结构化格式。

```python
        # 投资评级关键词
        self.rating_keywords = {
            '买入': ['买入', '强烈推荐', '推荐', 'Buy'],
            '增持': ['增持', '优于大市', 'Outperform'],
            '中性': ['中性', '持有', '大市同步', 'Hold'],
            '减持': ['减持', '弱于大市', 'Underperform', 'Sell']
        }
    
    def extract_metadata(self, text: str) -> ReportMetadata:
        """
        提取研报元数据
        :param text: 研报全文
        :return: 元数据对象
        """
        # 1. 提取标题（通常在第一行或封面）
        title = self._extract_title(text)
        
        # 2. 提取机构
        institution = self._extract_institution(text)
        
        # 3. 提取分析师
        analysts = self._extract_analysts(text)
        
        # 4. 提取日期
        publish_date = self._extract_date(text)
        
        # 5. 提取评级
        rating = self._extract_rating(text)
        
        # 6. 提取目标价
        target_price = self._extract_target_price(text)
        
        return ReportMetadata(
            title=title,
            institution=institution,
            analysts=analysts,
            publish_date=publish_date,
            rating=rating,
            target_price=target_price
        )
    
    def _extract_title(self, text: str) -> str:
        """提取标题"""
        lines = text.split('\n')
        # 标题通常在开头，且较短（<100 字符）
        for line in lines[:10]:
            line = line.strip()
            if 10 < len(line) < 100 and not line.endswith(('：', ':')):
                return line
        return lines[0].strip() if lines else ""
    
    def _extract_institution(self, text: str) -> str:
        """提取机构名称"""
        for inst in self.institutions:
            if inst in text[:500]:  # 机构名通常在开头 500 字
                return inst
        # 尝试正则匹配
        match = re.search(r'([^\s]+证券|[^\s]+研究所)', text[:500])
        if match:
            return match.group(1)
        return "未知机构"
    
    def _extract_analysts(self, text: str) -> List[str]:
        """提取分析师"""
        analysts = []
        # 常见格式："分析师：张三" 或 "证券分析师：李四"
        patterns = [
            r'分析师 [：:]\s*([^\n]+)',
            r'证券分析师 [：:]\s*([^\n]+)',
            r'报告作者 [：:]\s*([^\n]+)'
        ]
        for pattern in patterns:
            match = re.search(pattern, text[:1000])
            if match:
                raw = match.group(1)
                # 分割多个分析师
                analysts = re.split(r'[,,\s]+', raw.strip())
                break
        return analysts
    
    def _extract_date(self, text: str) -> str:
        """提取发布日期"""
        # 常见日期格式
        patterns = [
            r'(\d{4}年\d{1,2}月\d{1,2}日)',
            r'(\d{4}-\d{1,2}-\d{1,2})',
            r'(\d{4}/\d{1,2}/\d{1,2})'
        ]
        for pattern in patterns:
            match = re.search(pattern, text[:500])
            if match:
                return match.group(1)
        return ""
    
    def _extract_rating(self, text: str) -> Optional[str]:
        """提取投资评级"""
        for rating, keywords in self.rating_keywords.items():
            for keyword in keywords:
                if keyword in text[:1000]:  # 评级通常在摘要部分

        return None
    
    def _extract_target_price(self, text: str) -> Optional[float]:
        """提取目标价"""
        # 常见格式："目标价 XX 元" 或 "TP: XX"
        patterns = [
            r'目标价 [：:]?\s*(\d+(?:\.\d+)?)\s*元',
            r'TP[：:]?\s*(\d+(?:\.\d+)?)',
            r'目标价位 [：:]?\s*(\d+(?:\.\d+)?)'
        ]
        for pattern in patterns:
            match = re.search(pattern, text[:1000])
            if match:
                return float(match.group(1))
        return None
    
    def extract_financial_tables(self, tables: List[dict]) -> List[FinancialForecast]:

        results = []
        
        for table in tables:

            if self._is_financial_table(html):

    def _is_financial_table(self, html: str) -> bool:

        financial_keywords = [
```

#### M1.2 NLP 模型选型

| 方案 | 模型 | 优点 | 缺点 | 推荐度 |
|------|------|------|------|--------|
| **规则提取** | 关键词 + 正则 | 简单、快速、无需训练 | 覆盖率有限、灵活性差 | ⭐⭐⭐ |
| **抽取式摘要** | TextRank/BERT-Extract | 无需训练、效果稳定 | 无法生成新句子 | ⭐⭐⭐⭐ |
| **生成式摘要** | ChatGLM3-6B / Qwen-7B | 灵活、可生成连贯摘要 | 需 GPU、推理慢 | ⭐⭐⭐ |
| **混合方案** | 规则定位 + BERT 抽取 | 平衡效果与成本 | 实现复杂度中等 | ⭐⭐⭐⭐⭐ |

**推荐方案：混合方案（规则定位 + BERT 抽取）**

**理由：**
1. 规则定位可快速锁定核心观点章节（准确率>90%）
2. BERT 抽取保证摘要质量（ROUGE-L > 0.45）
3. 无需 GPU，CPU 可运行（推理速度~100ms/句）
4. 可离线部署，无 API 依赖

**技术实现：**
```python
from transformers import AutoModelForSequenceClassification, AutoTokenizer
import torch

class CoreViewExtractor:
    def __init__(self, model_name: str = 'bert-base-chinese'):
        # 加载预训练 BERT 模型（用于句子重要性评分）
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForSequenceClassification.from_pretrained(
            model_name, 
            num_labels=1  # 二分类：重要/不重要
        )
        # 使用中文财经领域微调的模型（可选）
        # model_name = 'shibing624/finetuned-chinese-bert'
    
    def extract_core_views(self, text: str, max_sentences: int = 5) -> List[str]:
        """
        提取核心观点
        :param text: 研报全文
        :param max_sentences: 最多返回的核心观点数量
        :return: 核心观点列表
        """
        # 1. 规则定位：找到核心观点章节
        core_section = self._locate_core_section(text)
        if not core_section:
            return []
        
        # 2. 句子分割
        sentences = self._split_sentences(core_section)
        if len(sentences) <= max_sentences:
            return sentences
        
        # 3. BERT 句子重要性评分
        scores = self._score_sentences(sentences)
        
        # 4. 选择 Top-N 句子
        top_indices = torch.topk(scores, min(max_sentences, len(scores))).indices
        core_views = [sentences[i] for i in sorted(top_indices.tolist())]
        
        return core_views
    
    def _locate_core_section(self, text: str) -> str:
        """定位核心观点章节"""
        keywords = ['核心观点', '投资摘要', '主要结论', 'Key Points', '投资要点']
        lines = text.split('\n')
        
        start_idx = -1
        for i, line in enumerate(lines[:30]):  # 前 30 行
            if any(kw in line for kw in keywords):
                start_idx = i
                break
        
        if start_idx == -1:
            return text[:2000]  # 默认取前 2000 字
        
        # 提取章节内容（到下一章节前）
        section_lines = []
        for line in lines[start_idx:start_idx + 50]:  # 最多 50 行
            if self._is_new_section(line):
                break
            section_lines.append(line)
        
        return '\n'.join(section_lines)
    
    def _score_sentences(self, sentences: List[str]) -> torch.Tensor:
        """BERT 句子重要性评分"""
        inputs = self.tokenizer(
            sentences,
            padding=True,
            truncation=True,
            max_length=128,
            return_tensors='pt'
        )
        
        with torch.no_grad():
            outputs = self.model(**inputs)
            scores = outputs.logits.squeeze(-1)
        
        # 位置加权：靠前的句子权重更高
        position_weights = torch.linspace(1.0, 0.5, len(sentences))
        scores = scores * position_weights
        
        return scores
```

#### M1.3 摘要质量评估指标

| 指标 | 定义 | 目标值 | 计算方式 |
|------|------|--------|---------|
| **召回率 (Recall)** | 提取的核心观点覆盖人工标注的比例 | ≥80% | `TP / (TP + FN)` |
| **精确率 (Precision)** | 提取的核心观点中正确的比例 | ≥85% | `TP / (TP + FP)` |
| **F1 分数** | 召回率与精确率的调和平均 | ≥82% | `2 * P * R / (P + R)` |
| **ROUGE-L** | 最长公共子序列相似度（与人工摘要对比） | ≥0.45 | `LCS / Reference_Length` |
| **人工满意度** | 人工评分（1-5 分） | ≥4.0 分 | 3 位标注员平均分 |

**评估流程：**

#### M2.2 跨页表格合并策略

**问题：** 大型表格跨越多个页面，需合并为完整表格。

**解决方案：**

```python
from dataclasses import dataclass
from typing import List, Optional

@dataclass
class TableFragment:
    """表格片段"""
    page_num: int
    html: str
    headers: List[str]  # 表头
    rows: List[List[str]]  # 数据行
    is_continued: bool  # 是否延续自上一页
    continuation_marker: str  # 续表标记（如"续表"、"(续)"）

class CrossPageTableMerger:
    def __init__(self):
        self.continuation_keywords = ['续表', '(续)', 'continued', '接上页']
    
    def merge_tables(self, fragments: List[TableFragment]) -> List[dict]:
        """
        合并跨页表格
        :param fragments: 表格片段列表（按页码排序）
        :return: 合并后的完整表格列表
        """
        if not fragments:
            return []
        
        merged_tables = []
        current_table = None
        
        for frag in fragments:
            if current_table is None:
                # 开始新表格
                current_table = {
                    'headers': frag.headers,
                    'rows': frag.rows.copy(),
                    'start_page': frag.page_num,
                    'end_page': frag.page_num
                }
            else:
                # 判断是否延续当前表格
                if self._is_continuation(current_table, frag):
                    # 合并
                    current_table['rows'].extend(frag.rows)
                    current_table['end_page'] = frag.page_num
                else:
                    # 保存当前表格，开始新表格
                    merged_tables.append(current_table)
                    current_table = {
                        'headers': frag.headers,
                        'rows': frag.rows.copy(),
                        'start_page': frag.page_num,
                        'end_page': frag.page_num
                    }
        
        # 保存最后一个表格
        if current_table:
            merged_tables.append(current_table)
        
        return merged_tables
    
    def _is_continuation(self, current_table: dict, frag: TableFragment) -> bool:
        """
        判断是否为表格延续
        :param current_table: 当前表格
        :param frag: 新片段
        :return: 是否延续
        """
        # 1. 检查续表标记
        if frag.is_continued:
            return True
        
        # 2. 检查表头是否一致
        if frag.headers == current_table['headers']:
            return True
        
        # 3. 检查列数是否一致
        if len(frag.rows) > 0 and len(current_table['rows']) > 0:
            if len(frag.rows[0]) == len(current_table['rows'][0]):
                return True
        
        return False
```

#### M2.3 表格结构还原方法

**解决方案：**

```python

        if not table:
            return html
        
        # 2. 检测合并单元格
        self._detect_merged_cells(table)
        
        # 3. 还原表头层级

            cells = row.find_all(['td', 'th'])
            for j, cell in enumerate(cells):

                colspan = cell.get('colspan', 1)
                
                if rowspan != 1 or colspan != 1:
                    # 标记为合并单元格
                    cell['data-merged'] = 'true'
                    cell['data-rowspan'] = rowspan
                    cell['data-colspan'] = colspan
    
    def _restore_header_hierarchy(self, table):
        """还原表头层级"""
        # 检测多级表头
        headers = table.find_all('th')
        
        # 根据内容判断层级
        for th in headers:
            text = th.get_text().strip()
            if any(kw in text for kw in ['合计', '总计', '平均', '占比']):
```

#### M3.2 准确率计算方法

**OCR 准确率：**
```python
def calculate_ocr_accuracy(ocr_result: str, ground_truth: str) -> float:
    """
    计算 OCR 识别准确率
    :param ocr_result: OCR 识别结果
    :param ground_truth: 人工标注标准答案
    :return: 准确率（0-1）
    """
    # 1. 字符级准确率
    from difflib import SequenceMatcher
    char_accuracy = SequenceMatcher(None, ocr_result, ground_truth).ratio()
    
    # 2. 词级准确率（中文分词后对比）
    import jieba
    ocr_words = set(jieba.cut(ocr_result))
    gold_words = set(jieba.cut(ground_truth))
    word_accuracy = len(ocr_words & gold_words) / len(ocr_words | gold_words)
    
    # 3. 数字准确率（单独计算，财报中数字更重要）
    import re
    ocr_numbers = set(re.findall(r'\d+(?:\.\d+)?%', ocr_result))
    gold_numbers = set(re.findall(r'\d+(?:\.\d+)?%', ground_truth))
    number_accuracy = len(ocr_numbers & gold_numbers) / len(gold_numbers) if gold_numbers else 1.0
    
    # 加权平均：数字权重 50%，字符 30%，词 20%
    final_accuracy = 0.5 * number_accuracy + 0.3 * char_accuracy + 0.2 * word_accuracy
    
    return final_accuracy
```

**研报提取准确率：**
```python
def calculate_report_extraction_accuracy(extracted: dict, gold: dict) -> dict:
    """
    计算研报提取准确率
    :param extracted: 模型提取结果
    :param gold: 人工标注标准答案
    :return: 各项指标准确率
    """
    metrics = {}
    
    # 1. 元数据准确率
    metrics['title'] = 1.0 if extracted.get('title') == gold.get('title') else 0.0
    metrics['institution'] = 1.0 if extracted.get('institution') == gold.get('institution') else 0.0
    metrics['rating'] = 1.0 if extracted.get('rating') == gold.get('rating') else 0.0
    metrics['target_price'] = 1.0 if extracted.get('target_price') == gold.get('target_price') else 0.0
    
    # 2. 核心观点 F1 分数
    metrics['core_views_f1'] = calculate_f1(
        extracted.get('core_views', []),
        gold.get('core_views', [])
    )
    
    # 3. 财务表格准确率
    metrics['table_accuracy'] = calculate_table_similarity(
        extracted.get('tables', []),
        gold.get('tables', [])
    )
    
    # 综合准确率（加权平均）
    weights = {
        'title': 0.1,
        'institution': 0.1,
        'rating': 0.15,
        'target_price': 0.15,
        'core_views_f1': 0.25,
        'table_accuracy': 0.25
    }
    metrics['overall'] = sum(metrics[k] * weights[k] for k in weights.keys())
    
    return metrics
```
