- **计算公式：** `worker_count = min(cpu_cores * 2, 16)`

```python
# 伪代码示例：流式解析避免内存溢出
def parse_pdf_streaming(pdf_path: str):
    """流式解析 PDF，逐页处理，避免一次性加载"""
    doc = fitz.open(pdf_path)
    for page_num in range(len(doc)):
        page = doc[page_num]
        # 提取当前页文本
        text = page.get_text()
        # 提取当前页表格
        tables = page.find_tables()
        # 即时处理并写入数据库
        yield {"page": page_num, "text": text, "tables": tables}
        # 释放当前页内存
        del page
    doc.close()
```

#### 3.2.1 单文件上限处理

**问题：** 单个 PDF 文件超过 500MB 如何处理？

**解决方案：**

```python
# 配置常量
MAX_SINGLE_FILE_SIZE = 200 * 1024 * 1024  # 200MB
MAX_PAGE_COUNT = 500  # 单文件最大页数

class PDFParser:
    def parse(self, file_path: str) -> ParseResult:
        file_size = os.path.getsize(file_path)
        
        # 检查文件大小
        if file_size > MAX_SINGLE_FILE_SIZE:
            # 策略 1：分块解析
            if self.config.enable_chunking:
                return self._parse_chunked(file_path)
            # 策略 2：拒绝并提示用户
            else:
                raise FileTooLargeError(
                    f"文件大小 {file_size}MB 超过上限 {MAX_SINGLE_FILE_SIZE}MB"
                )
        
        # 检查页数
        doc = fitz.open(file_path)
        if len(doc) > MAX_PAGE_COUNT:
            logger.warning(f"文件页数 {len(doc)} 超过建议值 {MAX_PAGE_COUNT}")
        
        return self._parse_normal(file_path)
    
    def _parse_chunked(self, file_path: str) -> ParseResult:
        """大文件分块解析"""
        doc = fitz.open(file_path)
        total_pages = len(doc)
        chunk_size = 100  # 每 100 页为一个块
        
        results = []
        for start_page in range(0, total_pages, chunk_size):
            end_page = min(start_page + chunk_size, total_pages)
            chunk_result = self._parse_page_range(doc, start_page, end_page)
            results.append(chunk_result)
            
            # 每处理一个块，保存一次检查点
            self._save_checkpoint(file_path, end_page)
        
        return self._merge_results(results)
```

**配置参数：**

```yaml
  max_file_size_mb: 200  # 单文件上限 200MB
  max_page_count: 500    # 单文件最大页数
  enable_chunking: true  # 启用大文件分块
  chunk_size_pages: 100  # 分块粒度（页数）
```

```python
# 任务提交控制
class TaskSubmitter:
    def __init__(self, max_concurrent_tasks=5):
        self.max_concurrent_tasks = max_concurrent_tasks
        self.redis_client = redis.Redis()
    
    def submit_task(self, task_data: dict) -> str:
        # 检查当前并发任务数
        current_tasks = self.redis_client.get('current_concurrent_tasks')
        if current_tasks and int(current_tasks) >= self.max_concurrent_tasks:
            # 加入等待队列
            return self._enqueue_waiting(task_data)
        
        # 提交任务
        task_id = self._submit_to_celery(task_data)
        self.redis_client.incr('current_concurrent_tasks')
        return task_id
    
    def _enqueue_waiting(self, task_data: dict) -> str:
        """加入等待队列"""
        waiting_queue = self.redis_client.lpush(
            'waiting_task_queue',
            json.dumps(task_data)
        )
        logger.info(f"任务加入等待队列，当前等待数：{self.redis_client.llen('waiting_task_queue')}")
        return waiting_queue
```

#### 3.2.3 网络重试机制

**问题：** MinIO 上传过程中网络中断如何恢复？

**解决方案：**

```python
import tenacity
from tenacity import retry, stop_after_attempt, wait_exponential, retry_if_exception_type

class StorageClient:
    @retry(
        stop=stop_after_attempt(3),  # 最多重试 3 次
        wait=wait_exponential(multiplier=1, min=2, max=30),  # 指数退避：2s, 4s, 8s
        retry=retry_if_exception_type((ConnectionError, TimeoutError)),
        reraise=True
    )
    def upload_to_minio(self, bucket: str, object_name: str, file_path: str) -> bool:
        """上传文件到 MinIO，支持自动重试"""
        try:
            self.minio_client.fput_object(
                bucket=bucket,
                object_name=object_name,
                file_path=file_path
            )
            logger.info(f"上传成功：{object_name}")
            return True
        except Exception as e:
            logger.error(f"上传失败：{object_name}, 错误：{e}")
            raise  # 触发重试
    
    def upload_with_checkpoint(self, bucket: str, object_name: str, file_path: str):
        """支持断点续传的上传"""
        uploaded_parts = self._get_uploaded_parts(bucket, object_name)
        total_size = os.path.getsize(file_path)
        part_size = 10 * 1024 * 1024  # 10MB per part
        
        with open(file_path, 'rb') as f:
            for part_num, offset in enumerate(range(0, total_size, part_size)):
                if offset in uploaded_parts:
                    logger.info(f"跳过已上传分片：{part_num}")
                    continue
                
                f.seek(offset)
                data = f.read(part_size)
                self._upload_part(bucket, object_name, part_num, data)
                self._save_upload_checkpoint(bucket, object_name, offset)
```

```python
            self.redis_client.hincrby(f'task:{task_id}:stats', 'failed', 1)
            # 记录错误样本
            self.redis_client.lpush(f'task:{task_id}:errors', json.dumps({
                'error': error_msg,
                'timestamp': datetime.now().isoformat()
            }))
        
        # 检查失败率
        self._check_failure_rate(task_id)
    
    def _check_failure_rate(self, task_id: str):
        """检查失败率是否超过阈值"""
        stats = self.redis_client.hgetall(f'task:{task_id}:stats')
        total = int(stats.get(b'total', 0))
        failed = int(stats.get(b'failed', 0))
        
        if total > 0:
            failure_rate = failed / total
            if failure_rate > self.failure_threshold:
                logger.error(f"任务 {task_id} 解析失败率 {failure_rate:.2%} 超过阈值 {self.failure_threshold:.2%}")
                self._trigger_alert(task_id, failure_rate)
    
    def _trigger_alert(self, task_id: str, failure_rate: float):
        """触发告警并暂停任务"""
        # 1. 暂停任务
        self.redis_client.set(f'task:{task_id}:paused', 'true')
        
        # 2. 发送告警
        alert_message = f"""
⚠️ 解析失败率告警
任务 ID: {task_id}
失败率：{failure_rate:.2%}
阈值：{self.failure_threshold:.2%}
已暂停任务，请人工介入检查
        """
        send_alert(alert_message)
        
        # 3. 保存错误样本供分析
        errors = self.redis_client.lrange(f'task:{task_id}:errors', 0, 99)
        save_error_samples(task_id, errors)
```

```python
class DataProcessor:
    def __init__(self, batch_size: int = 1000):
        self.batch_size = batch_size
        # 根据部署环境动态配置内存限制
        self.max_memory_mb = self._detect_memory_limit()
    
    def _detect_memory_limit(self) -> int:
        """检测容器/环境内存限制"""
        import psutil
        total_memory = psutil.virtual_memory().total / 1024 / 1024  # MB
        # 使用 50% 的总内存作为上限
        return int(total_memory * 0.5)
    
    def process_large_dataset(self, query: str):
        """分块查询，避免一次性加载大量数据"""
        offset = 0
        iteration = 0
        while True:
            chunk = db.query(
                query, 
                limit=self.batch_size, 
                offset=offset
            )
            if not chunk:
                break
            yield chunk
            offset += self.batch_size
            iteration += 1
            
            # 每 10 次迭代调用一次 GC，避免频繁 GC 影响性能
            if iteration % 10 == 0:
                gc.collect()
```

- **开发环境（32GB）：** 限制 16GB

- **生产环境（128GB）：** 限制 64GB
- **动态检测：** 根据容器 cgroup 限制自动调整

#### 4.4.2 评估代码

```python
class NL2SQLEvaluator:
    def __init__(self, db_connection):
        self.db = db_connection
    
    def evaluate(self, predictions: List[str], golds: List[str]) -> dict:
        """评估 NL2SQL 模型"""
        results = {
            'exact_match': 0,
            'execution_accuracy': 0,
            'syntax_correct': 0,
            'total': len(predictions)
        }
        
        for pred, gold in zip(predictions, golds):
            # 精确匹配
            if pred.strip() == gold.strip():
                results['exact_match'] += 1
            
            # 语法正确性检查
            if self._check_syntax(pred):
                results['syntax_correct'] += 1
            
            # 执行准确率检查
            if self._check_execution(pred, gold):
                results['execution_accuracy'] += 1
        
        return {
            'exact_match': results['exact_match'] / results['total'],
            'execution_accuracy': results['execution_accuracy'] / results['total'],
            'syntax_correct': results['syntax_correct'] / results['total'],
            'total_samples': results['total']
        }
    
    def _check_syntax(self, sql: str) -> bool:
        """检查 SQL 语法"""
        try:
            import sqlparse
            sqlparse.parse(sql)
            return True
        except:
            return False
    
    def _check_execution(self, pred: str, gold: str) -> bool:
        """检查执行结果"""
        try:
            pred_result = self.db.execute(pred)
            gold_result = self.db.execute(gold)
            return self._compare_results(pred_result, gold_result)
        except:
            return False
```

```python
def run_evaluation(model, test_data: List[dict]) -> EvaluationReport:
    """运行完整评估"""
    # 1. 自动评估
    predictions = []
    for sample in test_data:
        pred = model.generate(sample['question'], sample['schema'])
        predictions.append(pred)
    
    auto_metrics = evaluator.evaluate(predictions, [s['sql'] for s in test_data])
    
    # 2. 人工评估（抽样 500 条）
    sample_indices = random.sample(range(len(test_data)), 500)
    human_metrics = human_evaluator.evaluate(
        [test_data[i] for i in sample_indices],
        [predictions[i] for i in sample_indices]
    )
    
    # 3. 生成报告
    report = EvaluationReport(
        auto_metrics=auto_metrics,
        human_metrics=human_metrics,
        error_analysis=analyze_errors(predictions, test_data),
        recommendations=generate_recommendations(auto_metrics, human_metrics)
    )
    
    return report
```

```python
        @wraps(func)
        async def wrapper(current_user: dict = Depends(get_current_user), *args, **kwargs):

                    detail=f"缺少权限：{permission}"
                )
            
            return await func(current_user=current_user, *args, **kwargs)
```

```python
class AuditMiddleware:
    def __init__(self, app):
        self.app = app
    
    async def __call__(self, scope, receive, send):
        if scope['type'] != 'http':
            return await self.app(scope, receive, send)
        
        # 记录请求开始
        start_time = datetime.utcnow()
        
        # 执行请求
        response = await self.app(scope, receive, send)
        
        # 记录审计日志
        end_time = datetime.utcnow()
        duration_ms = (end_time - start_time).total_seconds() * 1000

            'action': scope['path'],
            'method': scope['method'],
            'response_status': response.get('status'),
            'response_time_ms': duration_ms,
            'ip_address': scope.get('client', ('',))[0]
        }))
```

## 八、备份恢复机制（新增 P1）

### 8.1 数据库备份

#### 8.1.1 定时备份脚本

```bash
#!/bin/bash
# scripts/backup_mysql.sh

BACKUP_DIR="/backup/mysql"
DATE=$(date +%Y%m%d_%H%M%S)
RETENTION_DAYS=30

# 全量备份
mysqldump \
    --host=mysql \
    --user=root \
    --password=${MYSQL_ROOT_PASSWORD} \
    --single-transaction \
    --routines \
    --triggers \
    caiwu_assistant | gzip > ${BACKUP_DIR}/full_${DATE}.sql.gz

# 删除旧备份
find ${BACKUP_DIR} -name "*.sql.gz" -mtime +${RETENTION_DAYS} -delete

# 上传到 MinIO（异地备份）
mc cp ${BACKUP_DIR}/full_${DATE}.sql.gz minio/backup-bucket/mysql/

echo "备份完成：full_${DATE}.sql.gz"
```

```bash
# crontab -e
```

## 十、风险评估与应对（修订版）

### 10.1 技术风险（修订）

| 风险项 | 概率 | 影响 | 应对措施（细化） |
|--------|------|------|-----------------|
| **PDF 解析失败率>5%** | 高 | 高 | **1. 多解析器 fallback**：PyMuPDF → pdfplumber → PyPDF2<br>**2. 建立失败样本库**：自动收集失败样本供分析<br>**3. 人工校验界面**：提供人工修正工具<br>**4. 自动告警**：失败率超阈值自动暂停并通知 |
| **NL2SQL 准确率<80%** | 高 | 高 | **1. 规则引擎兜底**：简单 query 用规则匹配<br>**2. 降级方案**：准确率不达标时切换为关键词检索<br>**3. 人工修正接口**：允许用户修正 SQL 并反馈<br>**4. 持续学习**：收集用户修正数据用于迭代训练 |
| **100G 数据性能不达标** | 中 | 高 | **1. 性能指标定义**：QPS>50、P95 延迟<2s<br>**2. 第 4 周开始压测**：早期发现瓶颈<br>**3. 预留 2 周优化缓冲**：测试阶段增加 1 周<br>**4. 水平扩展方案**：支持多 worker 节点 |
| **模型训练数据不足** | 中 | 中 | **1. 公开数据集**：Spider + BIRD (22K 条)<br>**2. 合成数据**：基于 schema 生成 (50K 条)<br>**3. 数据增强**：同义词替换、句式变换<br>**4. 迁移学习**：使用预训练 CodeLlama |
| **硬件资源不足** | 低 | 中 | **1. 云资源弹性**：支持按需扩展<br>**2. 资源监控**：Prometheus 实时监控<br>**3. 告警阈值**：80% 预警，90% 紧急<br>**4. 优化建议**：提供资源优化报告 |

## 十二、结论与建议

臣等谨呈上述技术方案（修订版），恭请圣裁。

### 核心建议：

1. **分阶段实施**：建议按六阶段推进，每阶段结束进行评审，确保方向正确
2. **敏捷迭代**：每两周一个 Sprint，及时交付可用功能，快速验证
